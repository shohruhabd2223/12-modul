Raqamni tozalash — eng ko‘p xato shu yerda

User +998 (90) 123-45-67 deb yozadi, provayder esa 998901234567 kutadi. Ikkovini bog‘laydigan 10 qatorlik funksiya.

accounts/phone.pynormalize_phone
import re

from rest_framework import serializers


def normalize_phone(raw):
    digits = re.sub(r"\D", "", str(raw))        # faqat raqamlar qoladi

    if len(digits) == 9:                        # 901234567 → 998901234567
        digits = "998" + digits

    if not re.fullmatch(r"998\d{9}", digits):
        raise serializers.ValidationError(
            "Raqam +998 XX XXX XX XX ko'rinishida bo'lishi kerak"
        )
    return digits

import logging

import requests
from django.conf import settings
from django.core.cache import cache
from rest_framework.exceptions import APIException

logger = logging.getLogger(__name__)


class SmsError(APIException):
    status_code = 503
    default_detail = "SMS yuborib bo'lmadi. Birozdan keyin urinib ko'ring."


def send_code(phone, code):
    backend = settings.SMS_BACKEND

    if backend == "console":
        print(f"[SMS] {phone} -> Tasdiqlash kodingiz: {code}")
    elif backend == "telegram":
        _telegram(phone, code)
    elif backend == "eskiz":
        _eskiz(phone, code)
    else:
        raise SmsError(f"Noma'lum SMS_BACKEND: {backend}")
accounts/sms.pyTelegram Gateway
def _telegram(phone, code):
    response = requests.post(
        "https://gatewayapi.telegram.org/sendVerificationMessage",
        headers={"Authorization": f"Bearer {settings.TELEGRAM_GATEWAY_TOKEN}"},
        json={"phone_number": f"+{phone}", "code": code, "ttl": 300},
        timeout=10,
    )
    data = response.json()
    if not data.get("ok"):
        logger.error("Telegram Gateway: %s", data.get("error"))
        raise SmsError()
accounts/sms.pyEskiz.uz
def _eskiz_token():
    token = cache.get("eskiz_token")
    if token:
        return token

    response = requests.post(
        "https://notify.eskiz.uz/api/auth/login",
        data={"email": settings.ESKIZ_EMAIL, "password": settings.ESKIZ_PASSWORD},
        timeout=10,
    )
    token = response.json()["data"]["token"]
    cache.set("eskiz_token", token, 60 * 60 * 24 * 20)   # ~20 kun
    return token


def _eskiz(phone, code):
    response = requests.post(
        "https://notify.eskiz.uz/api/message/sms/send",
        headers={"Authorization": f"Bearer {_eskiz_token()}"},
        data={
            "mobile_phone": phone,
            "message": f"Tasdiqlash kodingiz: {code}",
            "from": "4546",
        },
        timeout=10,
    )

    if response.status_code == 401:
        cache.delete("eskiz_token")       # token eskirgan — keyingi safar yangilanadi
        logger.error("Eskiz: token eskirgan")
        raise SmsError()

    if response.status_code >= 400:
        logger.error("Eskiz: %s", response.text)
        raise SmsError()

class RegisterView(generics.CreateAPIView):
    serializer_class = RegisterSerializer
    permission_classes = [AllowAny]

    def perform_create(self, serializer):
        user = serializer.save()
        try:
            send_verification_code(user)
        except SmsError:
            user.delete()      # SMS ketmadi — yarim qolgan akkaunt qolmasin
            raise


class VerifyPhoneView(generics.GenericAPIView):
    serializer_class = VerifyPhoneSerializer
    permission_classes = [AllowAny]

    def post(self, request):
        serializer = self.get_serializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        user = User.objects.filter(
            phone=serializer.validated_data["phone"], is_active=False
        ).first()
        record = PhoneCode.objects.filter(user=user).first()

        if record is None or record.is_expired():
            return Response({"detail": "Kod topilmadi yoki muddati tugagan"}, status=400)

        if record.attempts >= PhoneCode.MAX_ATTEMPTS:
            return Response({"detail": "Urinishlar tugadi, yangi kod so'rang"}, status=400)

        if record.code != serializer.validated_data["code"]:
            record.attempts += 1
            record.save()
            return Response({"detail": "Kod noto'g'ri"}, status=400)

        user.is_active = True
        user.save()
        record.delete()
        return Response({"detail": "Raqam tasdiqlandi. Endi login qiling."})
accounts/views.pyresend
RESEND_INTERVAL = timedelta(seconds=60)


class ResendCodeView(generics.GenericAPIView):
    serializer_class = ResendCodeSerializer
    permission_classes = [AllowAny]

    def post(self, request):
        serializer = self.get_serializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        user = User.objects.filter(
            phone=serializer.validated_data["phone"], is_active=False
        ).first()

        if user:
            record = PhoneCode.objects.filter(user=user).first()
            if record and timezone.now() - record.created_at < RESEND_INTERVAL:
                return Response({"detail": "1 daqiqada faqat 1 marta"}, status=429)
            send_verification_code(user)

        return Response({"detail": "Agar bunday raqam ro'yxatdan o'tgan bo'lsa, kod yuborildi"})
